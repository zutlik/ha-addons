#include <tunables/global>

profile claude-code flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities — AppArmor denies caps not listed here even when Docker grants them.
  # su/sudo need setuid+setgid+audit_write; PAM session setup needs dac_override/
  # dac_read_search; various bootstrap steps need chown/fowner/fsetid.
  capability setuid,
  capability setgid,
  capability audit_write,
  capability dac_override,
  capability dac_read_search,
  capability chown,
  capability fowner,
  capability fsetid,
  capability sys_resource,
  capability kill,

  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay / init
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  # tmux allocates pseudo-TTYs for the claude daemon and for attach clients
  # (see run.sh, claude-attach). Needs access to the ptmx/pts device pair.
  /dev/ptmx rw,
  /dev/pts/** rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Persistent add-on data (credentials, npm-global, remote_control_url.txt)
  /data/{,**} rw,

  # Shared volume (claude workspace)
  /share/{,**} rw,

  # Node.js and npm
  /usr/bin/node ix,
  /usr/bin/npm ix,
  /usr/bin/npx ix,

  # Bun runtime
  /root/.bun/{,**} rwix,

  # ttyd web terminal + its claude-user shell wrapper + tmux attach helper
  /usr/local/bin/ttyd ix,
  /usr/local/bin/claude-shell rix,
  /usr/local/bin/claude-attach rix,

  # Network (Anthropic API, Telegram, npm registry)
  network inet stream,
  network inet6 stream,

  # Required for npm and node operations
  /proc/*/status r,
  /proc/sys/kernel/random/uuid r,
  /etc/passwd r,
  /etc/resolv.conf r,
  /etc/ssl/certs/** r,
  /usr/share/ca-certificates/** r,
}
